const links = Array.from(document.querySelectorAll('a[href*="/productos/"]')); const procesados = new Set(); for (let a of links) { const href = a.getAttribute('href'); if (!href || href.endsWith('/productos/') || procesados.has(href)) continue; let card = a.closest('article') || a.closest('li') || a.parentElement; while (card && !card.innerText.includes('$') && card.parentElement && card.tagName !== 'BODY') { card = card.parentElement; } if (card && card.innerText.includes('$')) { procesados.add(href); const tituloEl = card.querySelector('h2, h3, h4') || a; // Búsqueda de EAN en código o texto visible let ean = ''; const match779 = card.innerHTML.match(/779\d{10}/); if (match779) { ean = match779[0]; } else { const matchGen = card.innerText.match(/(?:EAN|GTIN|CODIGO)[:\s\-]*([0-9]{8,14})/i); if (matchGen) ean = matchGen[1]; } const matchR = href.match(/R-(\d+)/); const codEnlace = matchR ? matchR[1] : ''; items.push({ nombre: tituloEl ? tituloEl.innerText.trim() : a.innerText.trim(), texto: card.innerText, ean: ean || codEnlace }); } } // Detección automática del número máximo de páginas let maxPag = 1; const pagBtns = Array.from(document.querySelectorAll('ul.pagination li, div[class*="pagination"] button, a[href*="page="]')); for (let btn of pagBtns) { const num = parseInt(btn.innerText.trim(), 10); if (!isNaN(num) && num > maxPag && num < 100) { maxPag = num; } } return { items, maxPag }; } """) if data and len(data["items"]) > 0: return data["items"], data["maxPag"] except Exception: if intento < reintentos_max: await asyncio.sleep(3) return [], 1 async def main(): print("Iniciando navegador Chromium...") async with async_playwright() as p: browser = await p.chromium.launch(headless=True) context = await browser.new_context( user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36", locale="es-AR", timezone_id="America/Argentina/Buenos_Aires" ) page = await context.new_page() total_crudo = [] for subcat in SUBCATEGORIAS: print(f"\nExplorando categoría: {subcat}") num_pag = 1 total_paginas_estimadas = 1 while num_pag <= total_paginas_estimadas: url = f"https://www.coto.com.ar/productos/{subcat}?page={num_pag}" items, max_pag = await extraer_articulos_con_reintento(page, url) if max_pag > total_paginas_estimadas: total_paginas_estimadas = max_pag if not items: print(f" Pág. {num_pag}: Sin productos.") else: print(f" Pág. {num_pag}/{total_paginas_estimadas}: {len(items)} productos.") total_crudo.extend(items) num_pag += 1 await asyncio.sleep(1) await browser.close() return total_crudo print("Iniciando extracción masiva de catálogo...") datos = asyncio.run(main()) # ============================================================================== # 3. PARSEO ESTRICTO: SOLO EAN, NOMBRE Y PRECIO DE VENTA FINAL # ============================================================================== catalogo_final = [] vistos = set() for item in datos: nombre = item['nombre'] if not nombre or nombre in vistos: continue vistos.add(nombre) # Filtrar solo la línea que contiene el precio directo (ignora x kilo / sin impuestos) lineas = item['texto'].split('\n') precio_num = 0.0 for linea in lineas: linea_limpia = linea.strip().lower() if any(descarte in linea_limpia for descarte in ["por 1", "por kilo", "sin impuestos", "regular"]): continue match = re.search(r'\$\s*([\d\.\,]+)', linea) if match: valor_str = match.group(1).replace('.', '').replace(',', '.') try: val = float(valor_str) if val > 0: precio_num = val break except ValueError: continue if precio_num <= 0: continue catalogo_final.append({ "ean": str(item['ean']).strip(), "nombre": nombre, "precio": precio_num }) print(f"\n==========================================") print(f"TOTAL ARTÍCULOS PROCESADOS: {len(catalogo_final)}") print(f"==========================================") with open("productos.json", "w", encoding="utf-8") as f: json.dump(catalogo_final, f, ensure_ascii=False, indent=2) print("Descargando productos.json...") files.download("productos.json")
